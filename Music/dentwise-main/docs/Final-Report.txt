4. White Box Testing
4.1 Selected Function: calculateFee
public double calculateFee(CustomerType type, double amount) {
if (amount <= 0) return 0.0;
switch (type) {
case VIP:
if (amount > 1000) return amount * 0.005;
return amount * 0.01;
case PREMIUM:
return amount * 0.015;
case REGULAR:
default:
if (amount < 50) return 1.0;
return amount * 0.02;
}
}
